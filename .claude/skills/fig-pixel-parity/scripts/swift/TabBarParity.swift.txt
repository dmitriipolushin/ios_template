import SwiftUI

@main
struct FigParityApp: App {
    var body: some Scene {
        WindowGroup {
            TabBar().onAppear(perform: snapshot)
        }
    }

    /// Writes the tab bar at @3x to Documents/tabbar.png for the pixel diff.
    @MainActor private func snapshot() {
        let renderer = ImageRenderer(content: TabBar())
        renderer.scale = 3
        guard let data = renderer.uiImage?.pngData() else { return }
        let url = URL.documentsDirectory.appending(path: "tabbar.png")
        try? data.write(to: url)
        print("SNAPSHOT \(url.path)")
    }
}

/// Design 06 · "Tab bar": 361x64, r26, white 82%, 1pt white stroke,
/// shadow #000 10% blur 28 y10, five 69x62 cells.
struct TabBar: View {
    var body: some View {
        HStack(spacing: 0) {
            Cell(icon: "Icons/plan", title: "Plan", active: true)
            Cell(icon: "Icons/shopping", title: "Shopping")
            ScanButton().frame(width: 68.6, height: 62)
            Cell(icon: "Icons/calories", title: "Calories")
            Cell(icon: "Icons/budget", title: "Budget")
        }
        .padding(.horizontal, 9)
        .frame(width: 361, height: 64)
        .background(
            RoundedRectangle(cornerRadius: 26)
                .fill(.white.opacity(0.82))
                .stroke(.white, lineWidth: 1)
                .shadow(color: .black.opacity(0.1), radius: 14, y: 10)
        )
    }
}

private struct Cell: View {
    let icon: String
    let title: String
    var active = false

    var body: some View {
        VStack(spacing: 3) {
            Image(icon).renderingMode(.template).frame(width: 22, height: 22)
            Text(title)
                .font(.custom("Inter", size: 12).weight(.medium))
                .tracking(-0.12)
                .frame(height: 18)
        }
        .foregroundStyle(active ? Color("Colors/ink") : Color("Colors/inkTertiary"))
        .frame(width: 68.6, height: 62)
    }
}

private struct ScanButton: View {
    var body: some View {
        Circle()
            // Figma blur r maps to SwiftUI radius r/2.
            .fill(Color("Colors/oliveDark")
                .shadow(.inner(color: Color(red: 0x4B/255, green: 0x56/255, blue: 0x1B/255), radius: 8.73, y: -29.7))
                .shadow(.inner(color: Color(red: 0x4B/255, green: 0x56/255, blue: 0x1B/255), radius: 9.75, y: -29.7))
                .shadow(.inner(color: .white.opacity(0.6), radius: 6.11, y: 2.96))
                .shadow(.drop(color: .black.opacity(0.1), radius: 8.13, y: 9.75)))
            .frame(width: 48, height: 48)
            .overlay(Image("Icons/scan").renderingMode(.template).foregroundStyle(.white))
    }
}
